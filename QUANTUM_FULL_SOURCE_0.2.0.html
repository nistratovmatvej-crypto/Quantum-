# Quantum

**Quantum Alpha 0.2** — native Android assistant foundation with a user-controlled microphone and speech-to-text.

## Current features
- First-launch local profile (device-only; not cloud authentication).
- Deep Atlantic teal / porcelain Quantum identity and concentric logo.
- Expandable in-app Dynamic Island pill. It is not a system-wide overlay.
- Microphone permission is requested only after the user taps the mic.
- Android speech-recognition service returns a transcript in the app.
- Listening stops when the user taps Stop, closes the panel, the app backgrounds, or the silence timer expires.
- Local model tier selector: Meteor / Comet / Nova. Selection is saved; model files are not bundled yet.
- Supernova Pro plan preview. Google Play billing is not configured; purchases are disabled.
- Settings persist locally.

## Important limitations
- The recognized text is shown, but no conversational AI provider is connected yet.
- Speech recognition availability and offline language packs depend on the Android device. The system recognition service may use the network.
- No local LLM is bundled; Meteor / Comet / Nova are selectable profiles only.
- The first-launch profile is local-only, not secure account authentication or cross-device sync.
- Subscription cards are informational only until Play Console products and billing integration are configured.
- Device-control actions, TTS, reminders, macros, widgets and notification summaries are not implemented yet.
- The Dynamic Island control is inside the app, not an always-on system overlay. The microphone is never always listening.

## Build and test
Requires JDK 17 and Android SDK platform 35.

```bash
./gradlew clean testDebugUnitTest
./gradlew assembleDebug
```

GitHub Actions runs the same unit-test and debug-APK build steps on pushes and pull requests. A green check means CI compiled the project, ran unit tests, and produced a debug APK; it does not replace device testing.

- **Timer honesty:** it counts as silence detection only if the mic really reports speech activity. With capture unavailable (now) LISTENING is a labelled **preview** ("No audio is captured"). A live mic without speech detection gets no timer, so it can never cut off a speaking user.

Written and expected to work: navigation, star/panel behaviour, settings persistence, device readout and tier verdicts. **Unverified until built.**
Incomplete by design: no real microphone, speech recognition, TTS, AI, memory, voices, billing, device actions, or system overlay.

class MainActivity : ComponentActivity() {
    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        enableEdgeToEdge()
        val app = application as QuantumApp
        setContent {
            QuantumTheme {
                val nav = rememberNavController()
                val settings by app.settings.settings.collectAsState()
                val start = if (app.profile.hasProfile()) "home" else "welcome"
                val homeVm: HomeViewModel = viewModel(factory = HomeViewModelFactory(app.micSession))
                Box(Modifier.fillMaxSize().background(QuantumColors.Background)) {
                    NavHost(nav, startDestination = start) {
                        composable("welcome") {
                            WelcomeScreen { displayName ->
                                app.profile.saveProfile(displayName)
                                nav.navigate("home") { popUpTo("welcome") { inclusive = true } }
                            }
                        }
                        composable("home") {
                            HomeScreen(
                                vm = homeVm,
                                settings = settings,
                                profileName = app.profile.displayName(),
                                onOpenSettings = { nav.navigate("settings") },
                                onOpenPlans = { nav.navigate("plans") }
                            )
                        }
                        composable("settings") {
                            SettingsScreen(app.settings, settings) { nav.popBackStack() }
                        }
                        composable("plans") {
                            PlansScreen(
                                selectedTier = settings.selectedTier,
                                onSelectTier = app.settings::setSelectedTier,
                                onBack = { nav.popBackStack() }
                            )
                        }
                    }
                }
            }
        }
    }

    override fun onStop() {
        val app = application as QuantumApp
        ViewModelProvider(this, HomeViewModelFactory(app.micSession))[HomeViewModel::class.java].panel.onClose()
        super.onStop()
    }

package com.quantum.assistant.audio

import android.Manifest
import android.content.Context
import android.content.Intent
import android.content.pm.PackageManager
import android.os.Bundle
import android.speech.RecognitionListener
import android.speech.RecognizerIntent
import android.speech.SpeechRecognizer
import androidx.core.content.ContextCompat
import java.util.Locale

sealed interface MicAvailability {
    data object Available : MicAvailability
    data class Unavailable(val reason: String) : MicAvailability
}

sealed interface MicStartResult {
    data object Started : MicStartResult
    data class Failed(val reason: String) : MicStartResult
}

interface MicListener {
    fun onSpeechDetected()
    fun onTranscript(text: String)
    fun onError(reason: String)
}

interface MicSession {
    val availability: MicAvailability
    val reportsSpeechActivity: Boolean
    fun start(listener: MicListener): MicStartResult
    fun stop()
}

/**
 * Uses Android's installed speech-recognition service. It starts only after the user
 * taps the microphone and grants RECORD_AUDIO. The system recognition service may
 * require internet depending on the device and installed language pack.
 */
class AndroidSpeechRecognizerSession(context: Context) : MicSession {
    private val appContext = context.applicationContext
    private var recognizer: SpeechRecognizer? = null
    private var active = false

    override val availability: MicAvailability =
        if (SpeechRecognizer.isRecognitionAvailable(appContext)) MicAvailability.Available
        else MicAvailability.Unavailable("На устройстве нет службы распознавания речи.")

    override val reportsSpeechActivity: Boolean = true

    override fun start(listener: MicListener): MicStartResult {
        if (availability !is MicAvailability.Available) {
            return MicStartResult.Failed((availability as MicAvailability.Unavailable).reason)
        }
        if (ContextCompat.checkSelfPermission(appContext, Manifest.permission.RECORD_AUDIO) != PackageManager.PERMISSION_GRANTED) {
            return MicStartResult.Failed("Нет разрешения на микрофон. Разреши доступ и попробуй снова.")
        }
        return try {
            stop()
            val speech = SpeechRecognizer.createSpeechRecognizer(appContext)
            recognizer = speech
            speech.setRecognitionListener(object : RecognitionListener {
                override fun onReadyForSpeech(params: Bundle?) { active = true }
                override fun onBeginningOfSpeech() { listener.onSpeechDetected() }
                override fun onRmsChanged(rmsdB: Float) {
                    if (rmsdB > 2.0f) listener.onSpeechDetected()
                }
                override fun onBufferReceived(buffer: ByteArray?) = Unit
                override fun onEndOfSpeech() = Unit
                override fun onError(error: Int) {
                    active = false
                    val message = when (error) {
                        SpeechRecognizer.ERROR_AUDIO -> "Ошибка аудиосистемы."
                        SpeechRecognizer.ERROR_INSUFFICIENT_PERMISSIONS -> "Android не выдал доступ к микрофону."
                        SpeechRecognizer.ERROR_NETWORK, SpeechRecognizer.ERROR_NETWORK_TIMEOUT -> "Служба распознавания недоступна. Проверь интернет."
                        SpeechRecognizer.ERROR_NO_MATCH, SpeechRecognizer.ERROR_SPEECH_TIMEOUT -> "Не удалось разобрать речь. Попробуй ещё раз."
                        SpeechRecognizer.ERROR_RECOGNIZER_BUSY -> "Служба распознавания занята."
                        else -> "Распознавание речи завершилось с ошибкой ($error)."
                    }
                    listener.onError(message)
                    stop()
                }
                override fun onResults(results: Bundle?) {
                    active = false
                    val text = results?.getStringArrayList(SpeechRecognizer.RESULTS_RECOGNITION)?.firstOrNull()
                    if (text.isNullOrBlank()) listener.onError("Речь не распознана. Попробуй ещё раз.")
                    else listener.onTranscript(text)
                    stop()
                }
                override fun onPartialResults(partialResults: Bundle?) = Unit
                override fun onEvent(eventType: Int, params: Bundle?) = Unit
            })
            val intent = Intent(RecognizerIntent.ACTION_RECOGNIZE_SPEECH).apply {
                putExtra(RecognizerIntent.EXTRA_LANGUAGE_MODEL, RecognizerIntent.LANGUAGE_MODEL_FREE_FORM)
                putExtra(RecognizerIntent.EXTRA_LANGUAGE, Locale.getDefault().toLanguageTag())
                putExtra(RecognizerIntent.EXTRA_PARTIAL_RESULTS, true)
                putExtra(RecognizerIntent.EXTRA_MAX_RESULTS, 3)
                putExtra(RecognizerIntent.EXTRA_CALLING_PACKAGE, appContext.packageName)
            }
            speech.startListening(intent)
            MicStartResult.Started
        } catch (t: Throwable) {
            stop()
            MicStartResult.Failed(t.message ?: "Не удалось запустить микрофон.")
        }
    }

    override fun stop() {
        val current = recognizer
        recognizer = null
        active = false
        if (current != null) {
            try { current.cancel() } catch (_: Throwable) { }
            try { current.destroy() } catch (_: Throwable) { }
        }
    }
}

/** Explicit fallback for tests or devices without a recognition service. */
object NoOpMicSession : MicSession {
    override val availability = MicAvailability.Unavailable("Распознавание речи недоступно на этом устройстве.")
    override val reportsSpeechActivity = false
    override fun start(listener: MicListener): MicStartResult =
        MicStartResult.Failed((availability as MicAvailability.Unavailable).reason)
    override fun stop() = Unit
}

package com.quantum.assistant.data

import android.content.SharedPreferences

/**
 * Local-only profile for the first launch. This is deliberately NOT presented
 * as a cloud account or secure authentication; sync/sign-in provider comes later.
 */
class ProfileStore(private val prefs: SharedPreferences) {
    fun hasProfile(): Boolean = !displayName().isNullOrBlank()
    fun displayName(): String? = prefs.getString(KEY_NAME, null)?.trim()?.takeIf { it.isNotEmpty() }
    fun saveProfile(name: String) {
        val clean = name.trim().take(MAX_NAME)
        require(clean.isNotEmpty()) { "Profile name must not be empty." }
        prefs.edit().putString(KEY_NAME, clean).apply()
    }
    fun clearProfile() { prefs.edit().remove(KEY_NAME).apply() }

    companion object {
        const val MAX_NAME = 32
        private const val KEY_NAME = "local_profile_name"
    }
}

package com.quantum.assistant.data

import android.content.SharedPreferences
import com.quantum.assistant.device.LocalTier

import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.asStateFlow

/** Memory modes from PRODUCT_SPEC. Alpha 0.1 only stores the choice. */
enum class MemoryMode(val label: String, val description: String) {
    OFF("Off", "No cross-session memory."),
    DIALOGUE_ONLY("Dialogue only", "Retain conversation history."),
    FULL("Full", "Retain useful facts and preferences."),
    FULL_CONFIGURABLE("Full + configurable", "Full memory with review, edit, pin and delete.")
}

data class AssistantSettings(
    val assistantName: String = DEFAULT_NAME,
    val speechSpeed: Float = 1.0f,
    val memoryMode: MemoryMode = MemoryMode.OFF,
    /** User's explicit local-tier choice; null = follow the (advisory) recommendation. No model is installed yet. */
    val selectedTier: LocalTier? = null
) {
    val displayName: String get() = assistantName.trim().ifEmpty { DEFAULT_NAME }

    companion object { const val DEFAULT_NAME = "Quantum" }
}

/** Tiny storage seam so persistence is unit-testable without Android. */
interface KeyValueStore {
    fun getString(key: String): String?
    fun putString(key: String, value: String)
}

class SharedPrefsStore(private val prefs: SharedPreferences) : KeyValueStore {
    override fun getString(key: String): String? = prefs.getString(key, null)
    override fun putString(key: String, value: String) { prefs.edit().putString(key, value).apply() }
}

/** Non-sensitive settings only. Reads once at construction; every change is written immediately. */
class SettingsRepository(private val store: KeyValueStore) {
    private val _settings = MutableStateFlow(load())
    val settings: StateFlow<AssistantSettings> = _settings.asStateFlow()

    private fun load(): AssistantSettings {
        val d = AssistantSettings()
        return AssistantSettings(
            assistantName = (store.getString(K_NAME) ?: d.assistantName).take(MAX_NAME),
            speechSpeed = (store.getString(K_SPEED)?.toFloatOrNull() ?: d.speechSpeed).coerceIn(MIN_SPEED, MAX_SPEED),
            memoryMode = MemoryMode.entries.firstOrNull { it.name == store.getString(K_MEMORY) } ?: d.memoryMode,
            selectedTier = LocalTier.entries.firstOrNull { it.name == store.getString(K_TIER) }
        )
    }

    fun setAssistantName(name: String) {
        val v = name.take(MAX_NAME)
        _settings.value = _settings.value.copy(assistantName = v); store.putString(K_NAME, v)
    }

    fun setSpeechSpeed(speed: Float) {
        val v = speed.coerceIn(MIN_SPEED, MAX_SPEED)
        _settings.value = _settings.value.copy(speechSpeed = v); store.putString(K_SPEED, v.toString())
    }

    fun setMemoryMode(mode: MemoryMode) {
        _settings.value = _settings.value.copy(memoryMode = mode); store.putString(K_MEMORY, mode.name)
    }

    fun setSelectedTier(tier: LocalTier?) {
        _settings.value = _settings.value.copy(selectedTier = tier); store.putString(K_TIER, tier?.name ?: "")
    }

    companion object {
        const val MAX_NAME = 30
        const val MIN_SPEED = 0.5f
        const val MAX_SPEED = 2.0f
        private const val K_NAME = "assistant_name"
        private const val K_SPEED = "speech_speed"
        private const val K_MEMORY = "memory_mode"
        private const val K_TIER = "selected_tier"
    }
}

package com.quantum.assistant.device

enum class LocalTier(val label: String) { METEOR("Quantum Meteor"), COMET("Quantum Comet"), NOVA("Quantum Nova") }

/** Raw facts about the device. Nullable fields mean "could not be read", never "zero". */
data class CapabilityInput(
    val totalRamMb: Long,
    val availableRamMb: Long,
    val sdkInt: Int,
    val abis: List<String>,
    /** android.os.PowerManager THERMAL_STATUS_* (0 none .. 6 shutdown); null below API 29 or if unreadable. */
    val thermalStatus: Int? = null,
    val batteryPercent: Int? = null,
    val powerSaveMode: Boolean? = null
)

enum class Verdict(val label: String) {
    MEETS("Meets requirements"), MARGINAL("Marginal"), BELOW("Below requirements")
}

data class TierAssessment(val tier: LocalTier, val verdict: Verdict, val reasons: List<String>)

data class CapabilityAssessment(
    val tiers: List<TierAssessment>,
    /** Highest tier that meets requirements, else Meteor if marginal, else null. Advisory only. */
    val recommended: LocalTier?,
    /** Signals that were readable and used. */
    val measured: List<String>,
    /** Signals NOT measured. Shown to the user so nothing is implied that wasn't checked. */
    val notMeasured: List<String>
)

/** Replaceable: a future implementation can check real model files, free storage and benchmark results. */
interface CapabilityAssessor {
    fun assess(input: CapabilityInput): CapabilityAssessment
}

package com.quantum.assistant.device

import android.app.ActivityManager
import android.content.Context
import android.os.BatteryManager
import android.os.Build
import android.os.PowerManager

/** Reads real values from Android. Anything unreadable becomes null rather than a guess. */
object DeviceSnapshotReader {
    fun read(context: Context): CapabilityInput {
        val am = context.getSystemService(Context.ACTIVITY_SERVICE) as ActivityManager
        val mem = ActivityManager.MemoryInfo().also { am.getMemoryInfo(it) }
        val pm = context.getSystemService(Context.POWER_SERVICE) as? PowerManager
        val bm = context.getSystemService(Context.BATTERY_SERVICE) as? BatteryManager
        val battery = bm?.getIntProperty(BatteryManager.BATTERY_PROPERTY_CAPACITY)?.takeIf { it in 0..100 }
        return CapabilityInput(
            totalRamMb = mem.totalMem / BYTES_PER_MB,
            availableRamMb = mem.availMem / BYTES_PER_MB,
            sdkInt = Build.VERSION.SDK_INT,
            abis = Build.SUPPORTED_ABIS.toList(),
            thermalStatus = if (Build.VERSION.SDK_INT >= 29) pm?.currentThermalStatus else null,
            batteryPercent = battery,
            powerSaveMode = pm?.isPowerSaveMode
        )
    }
