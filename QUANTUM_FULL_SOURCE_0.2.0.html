# Quantum

**Quantum Alpha 0.2** — native Android assistant foundation with a user-controlled microphone and speech-to-text.

## Current features
- First-launch local profile (device-only; not cloud authentication).
- Deep Atlantic teal / porcelain Quantum identity and concentric logo.
- Expandable in-app Dynamic Island pill. It is not a system-wide overlay.
- Microphone permission is requested only after the user taps the mic.
- Android speech-recognition service returns a transcript in the app.
- Listening stops when the user taps Stop, closes the panel, the app backgrounds, or the silence timer expires.
- Local model tier selector: Meteor / Comet / Nova. Selection is saved; model files are not bundled yet.
- Supernova Pro plan preview. Google Play billing is not configured; purchases are disabled.
- Settings persist locally.

## Important limitations
- The recognized text is shown, but no conversational AI provider is connected yet.
- Speech recognition availability and offline language packs depend on the Android device. The system recognition service may use the network.
- No local LLM is bundled; Meteor / Comet / Nova are selectable profiles only.
- The first-launch profile is local-only, not secure account authentication or cross-device sync.
- Subscription cards are informational only until Play Console products and billing integration are configured.
- Device-control actions, TTS, reminders, macros, widgets and notification summaries are not implemented yet.
- The Dynamic Island control is inside the app, not an always-on system overlay. The microphone is never always listening.

## Build and test
Requires JDK 17 and Android SDK platform 35.

```bash
./gradlew clean testDebugUnitTest
./gradlew assembleDebug
```

GitHub Actions runs the same unit-test and debug-APK build steps on pushes and pull requests. A green check means CI compiled the project, ran unit tests, and produced a debug APK; it does not replace device testing.

package com.quantum.assistant.audio

import android.Manifest
import android.content.Context
import android.content.Intent
import android.content.pm.PackageManager
import android.os.Bundle
import android.speech.RecognitionListener
import android.speech.RecognizerIntent
import android.speech.SpeechRecognizer
import androidx.core.content.ContextCompat
import java.util.Locale

sealed interface MicAvailability {
    data object Available : MicAvailability
    data class Unavailable(val reason: String) : MicAvailability
}

sealed interface MicStartResult {
    data object Started : MicStartResult
    data class Failed(val reason: String) : MicStartResult
}

interface MicListener {
    fun onSpeechDetected()
    fun onTranscript(text: String)
    fun onError(reason: String)
}

interface MicSession {
    val availability: MicAvailability
    val reportsSpeechActivity: Boolean
    fun start(listener: MicListener): MicStartResult
    fun stop()
}

/**
 * Uses Android's installed speech-recognition service. It starts only after the user
 * taps the microphone and grants RECORD_AUDIO. The system recognition service may
 * require internet depending on the device and installed language pack.
 */
class AndroidSpeechRecognizerSession(context: Context) : MicSession {
    private val appContext = context.applicationContext
    private var recognizer: SpeechRecognizer? = null
    private var active = false

    override val availability: MicAvailability =
        if (SpeechRecognizer.isRecognitionAvailable(appContext)) MicAvailability.Available
        else MicAvailability.Unavailable("На устройстве нет службы распознавания речи.")

    override val reportsSpeechActivity: Boolean = true

    override fun start(listener: MicListener): MicStartResult {
        if (availability !is MicAvailability.Available) {
            return MicStartResult.Failed((availability as MicAvailability.Unavailable).reason)
        }
        if (ContextCompat.checkSelfPermission(appContext, Manifest.permission.RECORD_AUDIO) != PackageManager.PERMISSION_GRANTED) {
            return MicStartResult.Failed("Нет разрешения на микрофон. Разреши доступ и попробуй снова.")
        }
        return try {
            stop()
            val speech = SpeechRecognizer.createSpeechRecognizer(appContext)
            recognizer = speech
            speech.setRecognitionListener(object : RecognitionListener {
                override fun onReadyForSpeech(params: Bundle?) { active = true }
                override fun onBeginningOfSpeech() { listener.onSpeechDetected() }
                override fun onRmsChanged(rmsdB: Float) {
                    if (rmsdB > 2.0f) listener.onSpeechDetected()
                }
                override fun onBufferReceived(buffer: ByteArray?) = Unit
                override fun onEndOfSpeech() = Unit
                override fun onError(error: Int) {
                    active = false
                    val message = when (error) {
                        SpeechRecognizer.ERROR_AUDIO -> "Ошибка аудиосистемы."
                        SpeechRecognizer.ERROR_INSUFFICIENT_PERMISSIONS -> "Android не выдал доступ к микрофону."
                        SpeechRecognizer.ERROR_NETWORK, SpeechRecognizer.ERROR_NETWORK_TIMEOUT -> "Служба распознавания недоступна. Проверь интернет."
                        SpeechRecognizer.ERROR_NO_MATCH, SpeechRecognizer.ERROR_SPEECH_TIMEOUT -> "Не удалось разобрать речь. Попробуй ещё раз."
                        SpeechRecognizer.ERROR_RECOGNIZER_BUSY -> "Служба распознавания занята."
                        else -> "Распознавание речи завершилось с ошибкой ($error)."
                    }
                    listener.onError(message)
                    stop()
                }
                override fun onResults(results: Bundle?) {
                    active = false
                    val text = results?.getStringArrayList(SpeechRecognizer.RESULTS_RECOGNITION)?.firstOrNull()
                    if (text.isNullOrBlank()) listener.onError("Речь не распознана. Попробуй ещё раз.")
                    else listener.onTranscript(text)
                    stop()
                }
                override fun onPartialResults(partialResults: Bundle?) = Unit
                override fun onEvent(eventType: Int, params: Bundle?) = Unit
            })
            val intent = Intent(RecognizerIntent.ACTION_RECOGNIZE_SPEECH).apply {
                putExtra(RecognizerIntent.EXTRA_LANGUAGE_MODEL, RecognizerIntent.LANGUAGE_MODEL_FREE_FORM)
                putExtra(RecognizerIntent.EXTRA_LANGUAGE, Locale.getDefault().toLanguageTag())
                putExtra(RecognizerIntent.EXTRA_PARTIAL_RESULTS, true)
                putExtra(RecognizerIntent.EXTRA_MAX_RESULTS, 3)
                putExtra(RecognizerIntent.EXTRA_CALLING_PACKAGE, appContext.packageName)
            }
            speech.startListening(intent)
            MicStartResult.Started
        } catch (t: Throwable) {
            stop()
            MicStartResult.Failed(t.message ?: "Не удалось запустить микрофон.")
        }
    }

    override fun stop() {
        val current = recognizer
        recognizer = null
        active = false
        if (current != null) {
            try { current.cancel() } catch (_: Throwable) { }
            try { current.destroy() } catch (_: Throwable) { }
        }
    }
}

/** Explicit fallback for tests or devices without a recognition service. */
object NoOpMicSession : MicSession {
    override val availability = MicAvailability.Unavailable("Распознавание речи недоступно на этом устройстве.")
    override val reportsSpeechActivity = false
    override fun start(listener: MicListener): MicStartResult =
        MicStartResult.Failed((availability as MicAvailability.Unavailable).reason)
    override fun stop() = Unit
}

package com.quantum.assistant.data

import android.content.SharedPreferences

/**
 * Local-only profile for the first launch. This is deliberately NOT presented
 * as a cloud account or secure authentication; sync/sign-in provider comes later.
 */
class ProfileStore(private val prefs: SharedPreferences) {
    fun hasProfile(): Boolean = !displayName().isNullOrBlank()
    fun displayName(): String? = prefs.getString(KEY_NAME, null)?.trim()?.takeIf { it.isNotEmpty() }
    fun saveProfile(name: String) {
        val clean = name.trim().take(MAX_NAME)
        require(clean.isNotEmpty()) { "Profile name must not be empty." }
        prefs.edit().putString(KEY_NAME, clean).apply()
    }
    fun clearProfile() { prefs.edit().remove(KEY_NAME).apply() }

    companion object {
        const val MAX_NAME = 32
        private const val KEY_NAME = "local_profile_name"
    }
}

data class AssistantSettings(

    val speechSpeed: Float = 1.0f,
    val memoryMode: MemoryMode = MemoryMode.OFF,
    /** User's explicit local-tier choice; null = follow the (advisory) recommendation. No model is installed yet. */

/** Tiny storage seam so persistence is unit-testable without Android. */
interface KeyValueStore {
    fun getString(key: String): String?
    fun putString(key: String, value: String)
}

class SharedPrefsStore(private val prefs: SharedPreferences) : KeyValueStore {
    override fun getString(key: String): String? = prefs.getString(key, null)
    override fun putString(key: String, value: String) { prefs.edit().putString(key, value).apply() }
}

/** Non-sensitive settings only. Reads once at construction; every change is written immediately. */
class SettingsRepository(private val store: KeyValueStore) {
    private val _settings = MutableStateFlow(load())
    val settings: StateFlow<AssistantSettings> = _settings.asStateFlow()

    private fun load(): AssistantSettings {
        val d = AssistantSettings()
        return AssistantSettings(
            assistantName = (store.getString(K_NAME) ?: d.assistantName).take(MAX_NAME),
            speechSpeed = (store.getString(K_SPEED)?.toFloatOrNull() ?: d.speechSpeed).coerceIn(MIN_SPEED, MAX_SPEED),
            memoryMode = MemoryMode.entries.firstOrNull { it.name == store.getString(K_MEMORY) } ?: d.memoryMode,
            selectedTier = LocalTier.entries.firstOrNull { it.name == store.getString(K_TIER) }
        )
    }

    fun setAssistantName(name: String) {
        val v = name.take(MAX_NAME)
        _settings.value = _settings.value.copy(assistantName = v); store.putString(K_NAME, v)
    }

    fun setSpeechSpeed(speed: Float) {
        val v = speed.coerceIn(MIN_SPEED, MAX_SPEED)
        _settings.value = _settings.value.copy(speechSpeed = v); store.putString(K_SPEED, v.toString())
    }

    fun setMemoryMode(mode: MemoryMode) {
        _settings.value = _settings.value.copy(memoryMode = mode); store.putString(K_MEMORY, mode.name)
    }

    fun setSelectedTier(tier: LocalTier?) {
        _settings.value = _settings.value.copy(selectedTier = tier); store.putString(K_TIER, tier?.name ?: "")
    }

    companion object {
        const val MAX_NAME = 30
        const val MIN_SPEED = 0.5f
        const val MAX_SPEED = 2.0f
        private const val K_NAME = "assistant_name"
        private const val K_SPEED = "speech_speed"
        private const val K_MEMORY = "memory_mode"
        private const val K_TIER = "selected_tier"
    }
}

    /** Signals NOT measured. Shown to the user so nothing is implied that wasn't checked. */

package com.quantum.assistant.device

import android.app.ActivityManager
import android.content.Context
import android.os.BatteryManager
import android.os.Build
import android.os.PowerManager

/** Reads real values from Android. Anything unreadable becomes null rather than a guess. */
object DeviceSnapshotReader {
    fun read(context: Context): CapabilityInput {
        val am = context.getSystemService(Context.ACTIVITY_SERVICE) as ActivityManager
        val mem = ActivityManager.MemoryInfo().also { am.getMemoryInfo(it) }
        val pm = context.getSystemService(Context.POWER_SERVICE) as? PowerManager
        val bm = context.getSystemService(Context.BATTERY_SERVICE) as? BatteryManager
        val battery = bm?.getIntProperty(BatteryManager.BATTERY_PROPERTY_CAPACITY)?.takeIf { it in 0..100 }
        return CapabilityInput(
            totalRamMb = mem.totalMem / BYTES_PER_MB,
            availableRamMb = mem.availMem / BYTES_PER_MB,
            sdkInt = Build.VERSION.SDK_INT,
            abis = Build.SUPPORTED_ABIS.toList(),

            batteryPercent = battery,

/**
 * Transparent rule-based assessor.
 * IMPORTANT: the numbers in [DEFAULT_REQUIREMENTS] are PLACEHOLDERS, not measurements of real model needs.
 * They must be replaced with figures from actual Meteor/Comet/Nova evaluation (roadmap Alpha 0.7).
 * This class does not and cannot measure GPU/NPU performance, storage, or inference speed.
 */
class ThresholdCapabilityAssessor(
    private val requirements: List<TierRequirements> = DEFAULT_REQUIREMENTS
) : CapabilityAssessor {

    data class TierRequirements(
        val tier: LocalTier, val minTotalRamMb: Long, val minAvailableRamMb: Long, val needs64Bit: Boolean
    )

    override fun assess(input: CapabilityInput): CapabilityAssessment {
        val tiers = requirements.map { assessTier(it, input) }
        val recommended = tiers.lastOrNull { it.verdict == Verdict.MEETS }?.tier
            ?: tiers.firstOrNull { it.verdict == Verdict.MARGINAL }?.tier
        val measured = mutableListOf("Total RAM", "Available RAM (changes constantly)", "Android version", "CPU ABI")
        val notMeasured = mutableListOf("GPU / NPU capability", "Free storage", "Real model speed or memory use")
        if (input.thermalStatus != null) measured += "Thermal status" else notMeasured += "Thermal status (not readable on this device)"
        if (input.batteryPercent != null) measured += "Battery level" else notMeasured += "Battery level (not readable)"
        if (input.powerSaveMode != null) measured += "Battery saver" else notMeasured += "Battery saver (not readable)"
        return CapabilityAssessment(tiers, recommended, measured, notMeasured)
    }

    private fun assessTier(r: TierRequirements, i: CapabilityInput): TierAssessment {
        val reasons = mutableListOf<String>()
        var verdict = Verdict.MEETS
        fun lower(to: Verdict, why: String) { if (to.ordinal > verdict.ordinal) verdict = to; reasons += why }

        if (i.totalRamMb < r.minTotalRamMb) lower(Verdict.BELOW, "Total RAM ${i.totalRamMb} MB < ${r.minTotalRamMb} MB")
        if (r.needs64Bit && i.abis.none { it == "arm64-v8a" || it == "x86_64" }) lower(Verdict.BELOW, "Needs a 64-bit CPU")
        if (verdict != Verdict.BELOW && i.availableRamMb < r.minAvailableRamMb)
            lower(Verdict.MARGINAL, "Only ${i.availableRamMb} MB RAM free now (wants ${r.minAvailableRamMb} MB)")
        if (verdict != Verdict.BELOW) {
            if ((i.thermalStatus ?: 0) >= THERMAL_SEVERE) lower(Verdict.MARGINAL, "Device is running hot")
            if (i.powerSaveMode == true) lower(Verdict.MARGINAL, "Battery saver is on")
            if ((i.batteryPercent ?: 100) in 0..14) lower(Verdict.MARGINAL, "Battery is low")
        }
        return TierAssessment(r.tier, verdict, reasons)
    }

    companion object {
        const val THERMAL_SEVERE = 3 // PowerManager.THERMAL_STATUS_SEVERE
        val DEFAULT_REQUIREMENTS = listOf(
            TierRequirements(LocalTier.METEOR, 2_500, 1_000, needs64Bit = false),
            TierRequirements(LocalTier.COMET, 5_000, 2_000, needs64Bit = true),
            TierRequirements(LocalTier.NOVA, 7_500, 3_500, needs64Bit = true)
        )
    }
}

enum class PanelState { COLLAPSED, EXPANDED_IDLE, LISTENING, PROCESSING, AWAITING_CONFIRMATION, SPEAKING, ERROR }

class PanelController(
    private val scope: CoroutineScope,
    private val mic: MicSession,
    private val timeoutSeconds: Int = 15
) {
    private val _state = MutableStateFlow(PanelState.COLLAPSED)
    val state: StateFlow<PanelState> = _state.asStateFlow()
    private val _secondsLeft = MutableStateFlow(timeoutSeconds)
    val secondsLeft: StateFlow<Int> = _secondsLeft.asStateFlow()
    private val _errorMessage = MutableStateFlow<String?>(null)
    val errorMessage: StateFlow<String?> = _errorMessage.asStateFlow()
    private val _transcript = MutableStateFlow<String?>(null)
    val transcript: StateFlow<String?> = _transcript.asStateFlow()

    val isCaptureLive: Boolean get() = mic.availability is MicAvailability.Available
    val timeoutIsSilenceDetection: Boolean get() = isCaptureLive && mic.reportsSpeechActivity
    val unavailableReason: String? get() = (mic.availability as? MicAvailability.Unavailable)?.reason
    private var timer: Job? = null

    private val micListener = object : MicListener {
        override fun onSpeechDetected() {
            if (_state.value == PanelState.LISTENING && mic.reportsSpeechActivity) restartTimer()
        }
        override fun onTranscript(text: String) {
            _transcript.value = text
            if (onTranscriptFinalized()) onResponseReady(speak = false)
        }
        override fun onError(reason: String) { onFailure(reason) }
    }

    fun onStarTap(): Boolean = move(PanelState.COLLAPSED, PanelState.EXPANDED_IDLE)

    fun onMicTap(): Boolean {
        if (_state.value != PanelState.EXPANDED_IDLE) return false
        _transcript.value = null
        if (isCaptureLive) {
            val result = try { mic.start(micListener) }
            catch (t: Throwable) { MicStartResult.Failed(t.message ?: "Microphone startup failed.") }
            if (result is MicStartResult.Failed) {
                safeStopMic()
                enter(PanelState.ERROR, result.reason)
                return true
            }
        } else {
            enter(PanelState.ERROR, unavailableReason ?: "Микрофон недоступен.")
            return true
        }
        enter(PanelState.LISTENING)
        if (mic.reportsSpeechActivity) restartTimer()
        return true
    }

    fun onStopListening(): Boolean {
        if (_state.value != PanelState.LISTENING) return false
        stopAudioAndTimer(); enter(PanelState.EXPANDED_IDLE); return true
    }

    fun onTranscriptFinalized(): Boolean {
        if (_state.value != PanelState.LISTENING) return false
        stopAudioAndTimer(); enter(PanelState.PROCESSING); return true
    }

    fun onActionProposed(): Boolean = move(PanelState.PROCESSING, PanelState.AWAITING_CONFIRMATION)
    fun onResponseReady(speak: Boolean): Boolean =
        move(PanelState.PROCESSING, if (speak) PanelState.SPEAKING else PanelState.EXPANDED_IDLE)
    fun onApprove(speak: Boolean): Boolean =
        move(PanelState.AWAITING_CONFIRMATION, if (speak) PanelState.SPEAKING else PanelState.EXPANDED_IDLE)
    fun onReject(): Boolean = move(PanelState.AWAITING_CONFIRMATION, PanelState.EXPANDED_IDLE)
    fun onSpeakingFinished(): Boolean = move(PanelState.SPEAKING, PanelState.EXPANDED_IDLE)

    fun onFailure(reason: String): Boolean {
        val s = _state.value
        if (s == PanelState.COLLAPSED || s == PanelState.ERROR) return false
        stopAudioAndTimer(); enter(PanelState.ERROR, reason); return true
    }

    fun onDismissError(): Boolean = move(PanelState.ERROR, PanelState.EXPANDED_IDLE)

    fun onClose(): Boolean {
        stopAudioAndTimer()
        if (_state.value == PanelState.COLLAPSED) return false
        enter(PanelState.COLLAPSED); return true
    }

    private fun move(from: PanelState, to: PanelState): Boolean {
        if (_state.value != from) return false
        enter(to); return true
    }

    private fun enter(to: PanelState, error: String? = null) {
        _errorMessage.value = if (to == PanelState.ERROR) error else null
        _state.value = to
    }

    private fun stopAudioAndTimer() {
        timer?.cancel(); timer = null
        _secondsLeft.value = timeoutSeconds
        safeStopMic()
    }

    private fun safeStopMic() {
        try { mic.stop() } catch (t: Throwable) {
            _errorMessage.value = t.message ?: "Microphone cleanup failed."
        }
    }

    private fun restartTimer() {
        timer?.cancel()
        timer = scope.launch {
            var left = timeoutSeconds
            while (left > 0) {
                _secondsLeft.value = left
                delay(1_000)
                left--
            }
            if (_state.value == PanelState.LISTENING) {
                stopAudioAndTimer()
                enter(PanelState.COLLAPSED)
            }
        }
    }
}

    override val reportsSpeechActivity: Boolean = true,
    private val failOnStart: String? = null

    var startCount = 0
    var listener: MicListener? = null

        failOnStart?.let { return MicStartResult.Failed(it) }

class PanelControllerTest {
    private enum class Ev(val fire: (PanelController) -> Boolean) {
        STAR({ it.onStarTap() }), MIC({ it.onMicTap() }), STOP({ it.onStopListening() }),
        TRANSCRIPT({ it.onTranscriptFinalized() }), PROPOSE({ it.onActionProposed() }),
        READY_SPEAK({ it.onResponseReady(true) }), READY_QUIET({ it.onResponseReady(false) }),
        APPROVE_SPEAK({ it.onApprove(true) }), APPROVE_QUIET({ it.onApprove(false) }), REJECT({ it.onReject() }),
        SPEAK_DONE({ it.onSpeakingFinished() }), FAIL({ it.onFailure("boom") }), DISMISS({ it.onDismissError() }), CLOSE({ it.onClose() })
    }

    private val valid: Map<Pair<PanelState, Ev>, PanelState> = mapOf(
        (PanelState.COLLAPSED to Ev.STAR) to PanelState.EXPANDED_IDLE,
        (PanelState.EXPANDED_IDLE to Ev.MIC) to PanelState.LISTENING,
        (PanelState.LISTENING to Ev.STOP) to PanelState.EXPANDED_IDLE,
        (PanelState.LISTENING to Ev.TRANSCRIPT) to PanelState.PROCESSING,
        (PanelState.PROCESSING to Ev.PROPOSE) to PanelState.AWAITING_CONFIRMATION,
        (PanelState.PROCESSING to Ev.READY_SPEAK) to PanelState.SPEAKING,
        (PanelState.PROCESSING to Ev.READY_QUIET) to PanelState.EXPANDED_IDLE,
        (PanelState.AWAITING_CONFIRMATION to Ev.APPROVE_SPEAK) to PanelState.SPEAKING,
        (PanelState.AWAITING_CONFIRMATION to Ev.APPROVE_QUIET) to PanelState.EXPANDED_IDLE,
        (PanelState.AWAITING_CONFIRMATION to Ev.REJECT) to PanelState.EXPANDED_IDLE,
        (PanelState.SPEAKING to Ev.SPEAK_DONE) to PanelState.EXPANDED_IDLE,
        (PanelState.ERROR to Ev.DISMISS) to PanelState.EXPANDED_IDLE
    ) + listOf(PanelState.EXPANDED_IDLE, PanelState.LISTENING, PanelState.PROCESSING,
        PanelState.AWAITING_CONFIRMATION, PanelState.SPEAKING, PanelState.ERROR).map { (it to Ev.CLOSE) to PanelState.COLLAPSED } +
        listOf(PanelState.EXPANDED_IDLE, PanelState.LISTENING, PanelState.PROCESSING,
            PanelState.AWAITING_CONFIRMATION, PanelState.SPEAKING).map { (it to Ev.FAIL) to PanelState.ERROR }

    private fun controller(scope: TestScope, mic: MicSession = FakeMic()) = PanelController(scope, mic)

    private fun PanelController.drive(target: PanelState): PanelController {
        when (target) {
            PanelState.COLLAPSED -> Unit
            PanelState.EXPANDED_IDLE -> onStarTap()
            PanelState.LISTENING -> { onStarTap(); onMicTap() }
            PanelState.PROCESSING -> { drive(PanelState.LISTENING); onTranscriptFinalized() }
            PanelState.AWAITING_CONFIRMATION -> { drive(PanelState.PROCESSING); onActionProposed() }
            PanelState.SPEAKING -> { drive(PanelState.PROCESSING); onResponseReady(true) }
            PanelState.ERROR -> { drive(PanelState.EXPANDED_IDLE); onFailure("boom") }
        }
        assertEquals(target, state.value)
        return this
    }

    @Test fun everyStateEventPairBehavesAsTableSays() {
        for (from in PanelState.values()) for (ev in Ev.values()) {
            val c = controller(TestScope()).drive(from)
            val expected = valid[from to ev]
            val accepted = ev.fire(c)
            assertEquals("$from + $ev accepted?", expected != null, accepted)
            assertEquals("$from + $ev", expected ?: from, c.state.value)
        }
    }

    @Test fun firstTapExpandsWithMicOff() {
        val mic = FakeMic(); val c = controller(TestScope(), mic)
        c.onStarTap()
        assertEquals(PanelState.EXPANDED_IDLE, c.state.value)
        assertEquals(0, mic.startCount); assertFalse(mic.running)
    }

    @Test fun micStartsOnlyOnSecondTap() {
        val mic = FakeMic(); val c = controller(TestScope(), mic)
        assertFalse(c.onMicTap()); assertEquals(0, mic.startCount)

    }

    }

    @Test fun closeCollapsesAndStopsMicFromEveryActiveState() {
        for (s in PanelState.values().filter { it != PanelState.COLLAPSED }) {
            val mic = FakeMic(); val c = controller(TestScope(), mic).drive(s)
            c.onClose()
            assertEquals(s.name, PanelState.COLLAPSED, c.state.value); assertFalse(s.name, mic.running)
        }
    }

    @Test fun unavailableMicrophoneDoesNotPretendToListen() {
        val c = PanelController(TestScope(), NoOpMicSession).drive(PanelState.EXPANDED_IDLE)
        c.onMicTap()
        assertEquals(PanelState.ERROR, c.state.value)
        assertFalse(c.isCaptureLive)
    }

    @Test fun speechActivityResetsTimeout() {
        val scope = TestScope(); val mic = FakeMic(); val c = PanelController(scope, mic).drive(PanelState.LISTENING)
        assertTrue(c.timeoutIsSilenceDetection)
        scope.runCurrent(); scope.advanceTimeBy(10_000)
        mic.listener!!.onSpeechDetected(); scope.runCurrent()
        scope.advanceTimeBy(10_000); scope.runCurrent()
        assertEquals(PanelState.LISTENING, c.state.value)
        scope.advanceTimeBy(6_000); scope.runCurrent()
        assertEquals(PanelState.COLLAPSED, c.state.value); assertFalse(mic.running)
    }

    @Test fun liveMicWithoutSpeechDetectionNeverTimesOut() {
        val scope = TestScope()
        val mic = object : MicSession {
            override val availability = MicAvailability.Available
            override val reportsSpeechActivity = false
            override fun start(listener: MicListener) = MicStartResult.Started
            override fun stop() = Unit
        }
        val c = controller(scope, mic).drive(PanelState.LISTENING)
        assertFalse(c.timeoutIsSilenceDetection)
        scope.advanceTimeBy(60_000); scope.runCurrent()
        assertEquals(PanelState.LISTENING, c.state.value)
    }

    @Test fun reEnteringListeningDoesNotStackTimers() {
        val scope = TestScope(); val mic = FakeMic()

        scope.runCurrent(); scope.advanceTimeBy(5_000); scope.runCurrent()
        c.onStopListening(); c.onMicTap(); scope.runCurrent()

        scope.advanceTimeBy(1_000); scope.runCurrent()
        assertEquals(PanelState.COLLAPSED, c.state.value)

    @Test fun micStartFailureGoesToErrorWithMessage() {
        val c = controller(TestScope(), FakeMic(failOnStart = "denied")).drive(PanelState.EXPANDED_IDLE)
        c.onMicTap()
        assertEquals(PanelState.ERROR, c.state.value); assertEquals("denied", c.errorMessage.value)
        c.onDismissError()
        assertEquals(PanelState.EXPANDED_IDLE, c.state.value); assertNull(c.errorMessage.value)
    }

    @Test fun thrownMicStartExceptionBecomesErrorState() {
        val mic = object : MicSession {
            override val availability = MicAvailability.Available
            override val reportsSpeechActivity = true
            override fun start(listener: MicListener): MicStartResult = error("permission bridge crashed")
            override fun stop() = Unit

        val c = controller(TestScope(), mic).drive(PanelState.EXPANDED_IDLE)
        assertTrue(c.onMicTap())
        assertEquals(PanelState.ERROR, c.state.value)
        assertEquals("permission bridge crashed", c.errorMessage.value)
    }

    @Test fun micErrorWhileListeningGoesToError() {
        val mic = FakeMic(); val c = controller(TestScope(), mic).drive(PanelState.LISTENING)
        mic.listener!!.onError("interrupted")
        assertEquals(PanelState.ERROR, c.state.value); assertFalse(mic.running)
    }

    @Test fun transcriptFromMicMovesToProcessingAndIsKept() {
        val mic = FakeMic(); val c = controller(TestScope(), mic).drive(PanelState.LISTENING)
        mic.listener!!.onTranscript("Привет, Quantum")
        assertEquals(PanelState.EXPANDED_IDLE, c.state.value); assertFalse(mic.running)
        assertEquals("Привет, Quantum", c.transcript.value)
    }

        assertTrue(NoOpMicSession.availability is MicAvailability.Unavailable)
        assertFalse(NoOpMicSession.reportsSpeechActivity)
    }

# Quantum — Technical Architecture

**Version:** Quantum Alpha 0.1  
**Status:** Proposed architecture; implementation decisions must be validated with Android prototypes.

## High-level modules

- app/ — application shell, navigation, lifecycle, dependency wiring.
- ui/ — Compose screens, design system, accessibility.
- floating/ — overlay button, expanded panel, animation and state machine.
- audio/ — microphone session, audio focus, speech activity/silence detection.
- speech/ — speech-to-text and text-to-speech adapters.
- assistant/ — conversation orchestration, intent interpretation, tool routing.
- ai/cloud/ — provider abstraction and cloud adapters.
- ai/local/ — local inference abstraction and model lifecycle.
- devicecontrol/ — allowlisted Android actions and confirmation workflow.
- memory/ — conversation history and optional long-term memory.
- voices/ — voice catalog, licensing metadata, downloads and queue.
- billing/ — Google Play Billing integration for Supernova.
- data/ — persistence, settings, secure storage and repositories.

## Core request flow

1. User taps the floating star to expand the panel.
2. User taps the panel microphone control.
3. Audio session requests/uses the required Android microphone permission and foreground execution where required.
4. Speech recognition produces a transcript.
5. Assistant orchestrator classifies the request: conversational response, cloud reasoning, local/offline command, or device action.
6. Device actions are converted into typed, allowlisted action proposals.
7. UI presents the exact action and asks for confirmation.
8. Only after confirmation does the action executor invoke the relevant Android API or supported intent.
9. Result is reported to the user and optionally added to conversation history according to memory settings.

## Floating panel state machine

States:
- COLLAPSED
- EXPANDED_IDLE
- LISTENING
- PROCESSING
- AWAITING_CONFIRMATION
- SPEAKING
- ERROR

Transitions:
- COLLAPSED + star tap -> EXPANDED_IDLE
- EXPANDED_IDLE + mic tap -> LISTENING
- LISTENING + transcript finalized -> PROCESSING
- LISTENING + 15 seconds silence -> stop capture -> COLLAPSED
- PROCESSING + action proposal -> AWAITING_CONFIRMATION
- AWAITING_CONFIRMATION + approve -> execute -> SPEAKING or EXPANDED_IDLE
- AWAITING_CONFIRMATION + reject -> EXPANDED_IDLE
- Any active state + explicit close -> stop audio safely -> COLLAPSED

The silence timer must reset when speech is detected. It must not cut off a user who is still speaking. The UI must expose a manual stop control.

## Android constraints

- Use Kotlin and Jetpack Compose for the native UI.
- Use Android-supported overlay and foreground-service patterns; do not assume an overlay can run invisibly or indefinitely.
- Request only necessary permissions, at the point of use.
- Do not use Accessibility Service as a blanket substitute for public Android APIs. If a feature genuinely requires accessibility, explain its purpose and obtain explicit user opt-in in system settings.
- Prefer public intents and supported APIs. Some system settings cannot be toggled directly by third-party apps.
- Respect Android background execution, microphone, notification, and foreground-service restrictions.

## Model runtime

Create a provider/runtime interface so Meteor, Comet, and Nova are selectable implementations rather than hard-coded UI labels.

Device recommendation should consider total and currently available memory, Android version and ABI, CPU/GPU/NPU support where available, thermal and battery constraints, model file size, and context requirements.

Model artifacts must be downloaded separately, checksum-verified, versioned, and accompanied by license metadata. Avoid committing large model binaries to Git.

## Cloud provider abstraction

Keep cloud provider credentials out of the Android client. Use a secure backend or a provider design that does not embed privileged API secrets in the APK. Define quotas and usage accounting server-side for free and Pro plans.

## Data and privacy

- Keep short-lived microphone audio in memory where possible; do not save raw recordings by default.
- Separate conversation history from long-term memory.
- Provide explicit controls to inspect, export, and delete stored memory/history.
- Store sensitive tokens using Android Keystore-backed mechanisms.
- Document what is sent to cloud providers before first use.

## Testing priorities

1. Overlay lifecycle and state transitions.
2. Microphone permission denial/revocation and interruption handling.
3. Silence timeout and speech-continuation behavior.
4. Confirmation gating for every device action.
5. Offline fallback and cloud-to-local transition.
6. Model memory pressure and thermal behavior.
7. Voice download queue recovery and checksum verification.
8. Billing entitlement validation and restore purchases.

# Quantum — Product Specification

**Version:** Quantum Alpha 0.1  
**Status:** Initial product baseline

## Identity

- Product name: Quantum
- Platform: Android
- Visual identity: deep navy background, simple white star with a subtle cool gradient and restrained glossy highlight; cyan/blue accents.
- Default assistant name: Quantum
- Default voice direction: calm, polished, Jarvis-inspired character; do not imply an official Marvel voice or use an unlicensed actor/character clone.
- Speech speed: 1.0x
- Communication style: friendly, concise, honest
- Language: follow Android device language by default; allow manual override.

## Floating assistant

1. A compact circular button displays the Quantum star.
2. First tap expands a horizontal floating panel. Microphone remains off.
3. Second tap inside the expanded panel starts microphone capture.
4. Speech is transcribed and sent through the assistant pipeline.
5. After 15 seconds with no detected speech, microphone capture stops and the panel collapses.
6. The user can manually stop listening or collapse the panel at any time.
7. The app must clearly indicate microphone state and respect Android foreground-service and overlay restrictions.

The 15-second timeout is a silence timeout, not a hard limit on the length of a spoken request.

## Assistant capabilities

Quantum is a full conversational AI and device assistant. It should understand intent, ask clarifying questions when needed, and use tools/actions when appropriate.

Potential Android actions include launching apps, opening supported settings screens, creating alarms/timers/reminders, and other capabilities permitted by Android APIs and granted permissions.

**Safety rule:** actions that change device state, access private data, send messages, place calls, delete files, or otherwise have meaningful consequences require a clear confirmation step. Permission prompts must be transparent and initiated only when the feature needs them.

## Intelligence modes

- Cloud AI is available to free and Pro users, subject to different usage limits.
- Local AI supports essential offline commands and lightweight requests.
- Quantum should preserve conversational context when switching between cloud and local modes where feasible.
- Network state and local-model availability should be visible to the user.

### Local model tiers

| Tier | Target device class | Intended trade-off |
|---|---|---|
| Quantum Meteor | Lower-resource devices | Small footprint and lower memory use; reduced reasoning and context capability |
| Quantum Comet | Mid-range devices | Balanced speed, memory use, and capability |
| Quantum Nova | High-end devices | Larger model and stronger local reasoning; higher RAM, storage, and thermal demand |

The app recommends a tier based on measured device capability and available resources, but lets the user choose another compatible tier. Do not promise that a tier will run based only on a chipset name.

## Memory

Four user-selectable modes:

1. Off — do not retain cross-session memory.
2. Dialogue only — retain conversation history.
3. Full — retain useful facts and preferences.
4. Full + configurable — full memory with review, edit, pin, and delete controls.

Recommend a mode based on device/storage capability, but keep memory controls separate from model selection. Memory should be stored independently of model weights and should not require keeping the model resident in RAM.

## Voice library

- 150 free base voices.
- 300 additional voices for Quantum Supernova Pro.
- Voices are presented as named character/persona profiles rather than generic labels where legally permitted.
- Voice language follows the app/device language by default; voice identity and accent are separate selectable properties.
- Users can preview voices before downloading.
- Users can select multiple voices and add them to a download queue.
- Queue supports progress, pause/resume where supported, retry, and cancellation.
- Voice assets are downloaded on demand, not bundled wholesale.

Only use voice assets with verified redistribution and commercial-use rights appropriate to the app. A voice being publicly downloadable or described as free does not establish that it is copyright-free. Character names, likenesses, and voice imitations may require separate permissions. Provide original or licensed alternatives when rights are unclear.

## Quantum Supernova Pro

- Monthly subscription.
- Annual purchase/subscription paid upfront at a lower effective monthly price.
- Same feature entitlement for monthly and annual plans.
- Higher cloud AI usage limits.
- 300 additional voice profiles.
- Further premium features may be added without removing core functionality from free users.

## Default settings

- Name: Quantum
- Voice: Jarvis-inspired (licensed/original voice only)
- Speech speed: 1.0x
- Tone: friendly, concise, honest
- Language: device language
- Floating assistant: manual activation only
- Microphone: off until the user taps the expanded panel

# Attempt to set APP_HOME

# Resolve links: $0 may be a link

# Need this for daisy-chained symlinks.
while
    APP_HOME=${app_path%"${app_path##*/}"}  # leaves a trailing /; empty if no leading path
    [ -h "$app_path" ]
do
    ls=$( ls -ld "$app_path" )
    link=${ls#*' -> '}
    case $link in             #(
      /*)   app_path=$link ;; #(
      *)    app_path=$APP_HOME$link ;;
    esac
done

# This is normally unused

# Discard cd standard output in case $CDPATH is set (https://github.com/gradle/gradle/issues/25036)

' "$PWD" ) || exit

# Use the maximum available, or set MAX_FD != -1 to use that value.
MAX_FD=maximum

warn () {
    echo "$*"
} >&2

die () {
    echo
    echo "$*"
    echo
    exit 1
} >&2

# OS specific support (must be 'true' or 'false').
cygwin=false
msys=false
darwin=false
nonstop=false
case "$( uname )" in                #(
  CYGWIN* )         cygwin=true  ;; #(
  Darwin* )         darwin=true  ;; #(
  MSYS* | MINGW* )  msys=true    ;; #(
  NONSTOP* )        nonstop=true ;;
esac

CLASSPATH=$APP_HOME/gradle/wrapper/gradle-wrapper.jar


# Determine the Java command to use to start the JVM.
if [ -n "$JAVA_HOME" ] ; then
    if [ -x "$JAVA_HOME/jre/sh/java" ] ; then
        # IBM's JDK on AIX uses strange locations for the executables
        JAVACMD=$JAVA_HOME/jre/sh/java
    else
        JAVACMD=$JAVA_HOME/bin/java
    fi
    if [ ! -x "$JAVACMD" ] ; then
        die "ERROR: JAVA_HOME is set to an invalid directory: $JAVA_HOME

Please set the JAVA_HOME variable in your environment to match the
location of your Java installation."
    fi
else
    JAVACMD=java
    if ! command -v java >/dev/null 2>&1
    then
        die "ERROR: JAVA_HOME is not set and no 'java' command could be found in your PATH.

Please set the JAVA_HOME variable in your environment to match the
location of your Java installation."
    fi
fi

# Increase the maximum file descriptors if we can.
if ! "$cygwin" && ! "$darwin" && ! "$nonstop" ; then
    case $MAX_FD in #(
      max*)
        # In POSIX sh, ulimit -H is undefined. That's why the result is checked to see if it worked.
        # shellcheck disable=SC2039,SC3045
        MAX_FD=$( ulimit -H -n ) ||
            warn "Could not query maximum file descriptor limit"
    esac
    case $MAX_FD in  #(
      '' | soft) :;; #(
      *)
        # In POSIX sh, ulimit -n is undefined. That's why the result is checked to see if it worked.
        # shellcheck disable=SC2039,SC3045
        ulimit -n "$MAX_FD" ||
            warn "Could not set maximum file descriptor limit to $MAX_FD"
    esac
fi

# Collect all arguments for the java command, stacking in reverse order:
#   * args from the command line
#   * the main class name
#   * -classpath
#   * -D...appname settings
#   * --module-path (only if needed)
#   * DEFAULT_JVM_OPTS, JAVA_OPTS, and GRADLE_OPTS environment variables.

# For Cygwin or MSYS, switch paths to Windows format before running java
if "$cygwin" || "$msys" ; then
    APP_HOME=$( cygpath --path --mixed "$APP_HOME" )
    CLASSPATH=$( cygpath --path --mixed "$CLASSPATH" )

    JAVACMD=$( cygpath --unix "$JAVACMD" )

    # Now convert the arguments - kludge to limit ourselves to /bin/sh
    for arg do
        if
            case $arg in                                #(
              -*)   false ;;                            # don't mess with options #(
              /?*)  t=${arg#/} t=/${t%%/*}              # looks like a POSIX filepath
                    [ -e "$t" ] ;;                      #(
              *)    false ;;
            esac
        then
            arg=$( cygpath --path --ignore --mixed "$arg" )
        fi
        # Roll the args list around exactly as many times as the number of
        # args, so each arg winds up back in the position where it started, but
        # possibly modified.
        #
        # NB: a `for` loop captures its iteration list before it begins, so
        # changing the positional parameters here affects neither the number of
        # iterations, nor the values presented in `arg`.
        shift                   # remove old arg
        set -- "$@" "$arg"      # push replacement arg
    done
fi


# Add default JVM options here. You can also use JAVA_OPTS and GRADLE_OPTS to pass JVM options to this script.
DEFAULT_JVM_OPTS='"-Xmx64m" "-Xms64m"'

# Collect all arguments for the java command:
#   * DEFAULT_JVM_OPTS, JAVA_OPTS, JAVA_OPTS, and optsEnvironmentVar are not allowed to contain shell fragments,
#     and any embedded shellness will be escaped.
#   * For example: A user cannot expect ${Hostname} to be expanded, as it is an environment variable and will be
#     treated as '${Hostname}' itself on the command line.

set -- \
        "-Dorg.gradle.appname=$APP_BASE_NAME" \
        -classpath "$CLASSPATH" \
        org.gradle.wrapper.GradleWrapperMain \
        "$@"

# Stop when "xargs" is not available.
if ! command -v xargs >/dev/null 2>&1
then
    die "xargs is not available"
fi

# Use "xargs" to parse quoted args.
#
# With -n1 it outputs one arg per line, with the quotes and backslashes removed.
#
# In Bash we could simply go:
#
#   readarray ARGS < <( xargs -n1 <<<"$var" ) &&
#   set -- "${ARGS[@]}" "$@"
#
# but POSIX shell has neither arrays nor command substitution, so instead we
# post-process each arg (as a line of input to sed) to backslash-escape any
# character that might be a shell metacharacter, then use eval to reverse
# that process (while maintaining the separation between arguments), and wrap
# the whole thing up as a single "set" statement.
#
# This will of course break if any of these variables contains a newline or
# an unmatched quote.
#

eval "set -- $(
        printf '%s\n' "$DEFAULT_JVM_OPTS $JAVA_OPTS $GRADLE_OPTS" |
        xargs -n1 |
        sed ' s~[^-[:alnum:]+,./:=@_]~\\&~g; ' |
        tr '\n' ' '
    )" '"$@"'

exec "$JAVACMD" "$@"
