package com.quantum.assistant.ui

import androidx.compose.foundation.Canvas
import androidx.compose.ui.Modifier
import androidx.compose.runtime.Composable
import androidx.compose.ui.geometry.Offset
import androidx.compose.ui.graphics.Brush
import androidx.compose.ui.graphics.Color
import androidx.compose.ui.graphics.drawscope.Stroke
import com.quantum.assistant.ui.theme.QuantumColors

/** Geometric concentric Quantum mark. Drawn natively for crisp adaptive sizing. */
@Composable
fun QuantumStar(modifier: Modifier = Modifier, glow: Boolean = false) {
    Canvas(modifier) {
        val c = center
        val r = size.minDimension * .46f
        if (glow) drawCircle(
            brush = Brush.radialGradient(
                listOf(QuantumColors.Accent.copy(alpha = .22f), Color.Transparent),
                center = c, radius = r * 1.55f
            ), radius = r * 1.55f, center = c
        )
        drawCircle(QuantumColors.Porcelain, r, c)
        drawCircle(
            brush = Brush.linearGradient(
                listOf(QuantumColors.AccentBright, QuantumColors.AccentDeep),
                start = Offset(c.x - r, c.y - r), end = Offset(c.x + r, c.y + r)
            ), r * .62f, c
        )
        drawCircle(QuantumColors.Porcelain, r * .19f, c)
        drawCircle(Color.White.copy(alpha = .18f), r * .985f, c,
            style = Stroke(width = (r * .012f).coerceAtLeast(1f)))
    }
}

class PanelActions(
    val onStarTap: () -> Unit,
    val onMicTap: () -> Unit,
    val onStopListening: () -> Unit,
    val onClose: () -> Unit,
    val onApprove: () -> Unit,
    val onReject: () -> Unit,
    val onSpeakingFinished: () -> Unit,
    val onDismissError: () -> Unit
) {
    companion object {
        fun from(c: PanelController, onMicTap: () -> Unit = { c.onMicTap() }) = PanelActions(
            { c.onStarTap() }, onMicTap, { c.onStopListening() }, { c.onClose() },
            { c.onApprove(speak = false) }, { c.onReject() }, { c.onSpeakingFinished() }, { c.onDismissError() }
        )
    }
}

@Composable
fun FloatingAssistant(
    state: PanelState,
    secondsLeft: Int,
    captureLive: Boolean,
    timeoutIsSilenceDetection: Boolean,
    errorMessage: String?,
    actions: PanelActions,
    modifier: Modifier = Modifier
) {
    val expanded = state != PanelState.COLLAPSED
    val listening = state == PanelState.LISTENING
    val shape = RoundedCornerShape(34.dp)

    Row(
        modifier = modifier.animateContentSize(spring(dampingRatio = 0.85f, stiffness = 400f))
            .background(QuantumColors.BackgroundElevated, shape)
            .border(1.dp, if (listening) QuantumColors.AccentBright else QuantumColors.Outline, shape)
            .padding(4.dp),
        verticalAlignment = Alignment.CenterVertically,
        horizontalArrangement = Arrangement.spacedBy(4.dp)
    ) {
        Box(
            Modifier.size(60.dp).clickable(enabled = !expanded, role = Role.Button, onClick = actions.onStarTap)
                .semantics { contentDescription = if (expanded) "Quantum" else "Открыть Quantum" },
            contentAlignment = Alignment.Center
        ) { QuantumStar(Modifier.size(48.dp), glow = !expanded) }

        if (expanded) {
            val (title, detail) = when (state) {
                PanelState.EXPANDED_IDLE -> "Готов к работе" to "Нажми на микрофон, чтобы начать."
                PanelState.LISTENING -> if (timeoutIsSilenceDetection) {
                    "Слушаю тебя" to "Остановка при тишине: $secondsLeft сек."
                } else "Слушаю" to "Нажми стоп, когда закончишь."
                PanelState.PROCESSING -> "Обрабатываю" to "Распознаю речь."
                PanelState.AWAITING_CONFIRMATION -> "Нужно подтверждение" to "Ничего не выполнится без тебя."
                PanelState.SPEAKING -> "Говорю" to ""
                PanelState.ERROR -> "Не удалось запустить" to (errorMessage ?: "Проверь разрешение микрофона.")
                PanelState.COLLAPSED -> "" to ""
            }
            Column(
                Modifier.widthIn(min = 126.dp, max = 196.dp).padding(horizontal = 4.dp)
                    .semantics { liveRegion = LiveRegionMode.Polite }
            ) {
                Text(title, style = MaterialTheme.typography.labelLarge, color = if (listening) QuantumColors.AccentBright else QuantumColors.Porcelain)
                if (detail.isNotEmpty()) Text(detail, style = MaterialTheme.typography.bodySmall, color = QuantumColors.TextMuted)
                when (state) {
                    PanelState.AWAITING_CONFIRMATION -> Row {
                        TextButton(onClick = actions.onApprove) { Text("Да") }
                        TextButton(onClick = actions.onReject) { Text("Нет") }
                    }
                    PanelState.SPEAKING -> TextButton(onClick = actions.onSpeakingFinished) { Text("Стоп") }
                    PanelState.ERROR -> TextButton(onClick = actions.onDismissError) { Text("Понятно") }
                    else -> Unit
                }
            }
            if (state == PanelState.EXPANDED_IDLE || listening) {
                MicButton(listening, if (listening) actions.onStopListening else actions.onMicTap)
            }
            Box(
                Modifier.size(46.dp).clickable(role = Role.Button, onClick = actions.onClose)
                    .semantics { contentDescription = "Закрыть панель" },
                contentAlignment = Alignment.Center
            ) { Icon(Icons.Default.Close, null, tint = QuantumColors.TextMuted) }

    FULL_CONFIGURABLE("Full + configurable", "Full memory with review, edit, pin and delete.")
}

data class AssistantSettings(
    val assistantName: String = DEFAULT_NAME,
    val speechSpeed: Float = 1.0f,
    val memoryMode: MemoryMode = MemoryMode.OFF,
    /** User's explicit local-tier choice; null = follow the (advisory) recommendation. No model is installed yet. */
    val selectedTier: LocalTier? = null
) {
    val displayName: String get() = assistantName.trim().ifEmpty { DEFAULT_NAME }

    companion object { const val DEFAULT_NAME = "Quantum" }
}

/** Tiny storage seam so persistence is unit-testable without Android. */
interface KeyValueStore {
    fun getString(key: String): String?
    fun putString(key: String, value: String)
}

class SharedPrefsStore(private val prefs: SharedPreferences) : KeyValueStore {
    override fun getString(key: String): String? = prefs.getString(key, null)
    override fun putString(key: String, value: String) { prefs.edit().putString(key, value).apply() }
}

/** Non-sensitive settings only. Reads once at construction; every change is written immediately. */
class SettingsRepository(private val store: KeyValueStore) {
    private val _settings = MutableStateFlow(load())
    val settings: StateFlow<AssistantSettings> = _settings.asStateFlow()

    private fun load(): AssistantSettings {
        val d = AssistantSettings()
        return AssistantSettings(
            assistantName = (store.getString(K_NAME) ?: d.assistantName).take(MAX_NAME),
            speechSpeed = (store.getString(K_SPEED)?.toFloatOrNull() ?: d.speechSpeed).coerceIn(MIN_SPEED, MAX_SPEED),
            memoryMode = MemoryMode.entries.firstOrNull { it.name == store.getString(K_MEMORY) } ?: d.memoryMode,
            selectedTier = LocalTier.entries.firstOrNull { it.name == store.getString(K_TIER) }
        )
    }

    fun setAssistantName(name: String) {
        val v = name.take(MAX_NAME)
        _settings.value = _settings.value.copy(assistantName = v); store.putString(K_NAME, v)
    }

    fun setSpeechSpeed(speed: Float) {
        val v = speed.coerceIn(MIN_SPEED, MAX_SPEED)
        _settings.value = _settings.value.copy(speechSpeed = v); store.putString(K_SPEED, v.toString())
    }

    fun setMemoryMode(mode: MemoryMode) {
        _settings.value = _settings.value.copy(memoryMode = mode); store.putString(K_MEMORY, mode.name)
    }

    fun setSelectedTier(tier: LocalTier?) {
        _settings.value = _settings.value.copy(selectedTier = tier); store.putString(K_TIER, tier?.name ?: "")
    }

    companion object {
        const val MAX_NAME = 30
        const val MIN_SPEED = 0.5f
        const val MAX_SPEED = 2.0f
        private const val K_NAME = "assistant_name"
        private const val K_SPEED = "speech_speed"
        private const val K_MEMORY = "memory_mode"
        private const val K_TIER = "selected_tier"
    }
}
